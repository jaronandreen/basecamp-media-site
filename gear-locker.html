// ── HELPERS ───────────────────────────────────────────────────────────────

function stars(rating) {

  const n = parseInt(rating) || 0;

  return Array.from({length:5}, (_,i) =>

    `<span class="star ${i < n ? '' : 'empty'}">${i < n ? '★' : '☆'}</span>`

  ).join('');

}


function linkBtns(aff, amz, ltk, size='sm') {

  const cls = size === 'sm' ? 'gear-link-sm' : 'link-btn';

  let html = '';

  if (aff) html += `<a href="${aff}" target="_blank" rel="nofollow" class="${cls} p">Shop ↗</a>`;

  if (amz) html += `<a href="${amz}" target="_blank" rel="nofollow" class="${cls}">Amazon ↗</a>`;

  if (ltk) html += `<a href="${ltk}" target="_blank" rel="nofollow" class="${cls}">LTK ↗</a>`;

  if (!aff && !amz && !ltk) html += `<span class="${cls}" style="opacity:0.3;cursor:default">Link coming</span>`;

  return html;

}


function statusClass(status) {

  if (!status) return '';

  const s = status.toLowerCase();

  if (s === 'wishlist') return 'wishlist';

  if (s === 'retired')  return 'retired';

  return '';

}


// ── RENDER FEATURED ───────────────────────────────────────────────────────

// ── RENDER GEAR GRID ──────────────────────────────────────────────────────

let allItems = [];

let activeFilter = 'All';


function renderGear(filter) {

  const container = document.getElementById('gear-container');

  activeFilter = filter;


  const filtered = filter === 'All'

    ? allItems

    : allItems.filter(r => r[COL.category] === filter);


  // Group by category

  const byCategory = {};

  filtered.forEach(r => {

    const cat = r[COL.category] || 'Other';

    if (!byCategory[cat]) byCategory[cat] = [];

    byCategory[cat].push(r);

  });


  if (!Object.keys(byCategory).length) {

    container.innerHTML = `<div class="loading">No items found</div>`;

    return;

  }


  container.innerHTML = Object.entries(byCategory).map(([cat, items]) => `

    <div class="cat-section fade-in">

      <div class="cat-header">

        <span class="cat-title">${CAT_ICONS[cat] || '🎒'} ${cat}</span>

        <span class="cat-count">${items.length} item${items.length !== 1 ? 's' : ''}</span>

      </div>

      <div class="gear-grid">

        ${items.map(r => `

          <div class="gear-card">

            <div class="gear-card-top">

              <span class="gear-cat-tag">${r[COL.subcat] || r[COL.category]}</span>

              <span class="status-dot ${statusClass(r[COL.status])}" title="${r[COL.status]}"></span>

            </div>

            <div class="gear-name">${r[COL.name]}</div>

            <div class="gear-brand-sm">${r[COL.brand]}</div>

            ${r[COL.rating] ? `<div class="gear-stars">${stars(r[COL.rating])}</div>` : ''}

            <div class="gear-take">${(r[COL.take] || '').substring(0,120)}${(r[COL.take]||'').length > 120 ? '…' : ''}</div>

            <div class="gear-links-sm">${linkBtns(r[COL.affiliate], r[COL.amazon], r[COL.ltk], 'sm')}</div>

          </div>

        `).join('')}

      </div>

    </div>

  `).join('');

}


// ── RENDER FILTERS ────────────────────────────────────────────────────────

function renderFilters(items) {

  const cats = [...new Set(items.map(r => r[COL.category]).filter(Boolean))].sort();

  const row  = document.getElementById('filter-row');

  cats.forEach(cat => {

    const btn = document.createElement('button');

    btn.className = 'filter-btn';

    btn.textContent = `${CAT_ICONS[cat] || ''} ${cat}`;

    btn.onclick = () => filterGear(cat, btn);

    row.appendChild(btn);

  });

}


function filterGear(cat, btn) {

  document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));

  btn.classList.add('active');

  renderGear(cat);

}


// ── RENDER STATS ──────────────────────────────────────────────────────────

function renderStats(items) {

  const current  = items.filter(r => r[COL.status] === 'Current');

  const featured = items.filter(r => r[COL.featured] === 'Y');

  const cats     = new Set(items.map(r => r[COL.category]).filter(Boolean));

  document.getElementById('stat-total').textContent    = current.length;

  document.getElementById('stat-featured').textContent = featured.length;

  document.getElementById('stat-cats').textContent     = cats.size;

}

// ── FETCH FROM GOOGLE SHEETS ──────────────────────────────────────────────

async function loadGear() {

  try {

    const res  = await fetch(API_URL);

    const data = await res.json();

    if (!data.values) throw new Error('No data');


    allItems = data.values.filter(r => r[COL.name]);


    renderStats(allItems);

    renderFeatured(allItems);

    renderFilters(allItems);

    renderGear('All');

    renderBundles();

  } catch(err) {

    console.error('Gear load error:', err);

    // Fallback message if API key not set yet

    document.getElementById('featured-container').innerHTML = `

      <div class="loading" style="flex-direction:column;gap:0.5rem;text-align:center;padding:3rem">

        <span style="font-size:1.5rem">⚙️</span>

        <span>Add your Sheet ID and API Key to activate the live feed</span>

        <span style="opacity:0.4;font-size:0.6rem">See the SHEET_ID and API_KEY variables at the top of this file</span>

      </div>`;

    document.getElementById('gear-container').innerHTML = '';

    renderBundles();

  }

}
